cmake_minimum_required(VERSION 3.20)
project(BabelPolicy C)
add_library(babel_policy SHARED BabelPolicy.c OcrPixels.c)
target_include_directories(babel_policy PUBLIC include)
set_target_properties(babel_policy PROPERTIES C_STANDARD 11 C_VISIBILITY_PRESET hidden)
if(MSVC)
  set_property(TARGET babel_policy PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(babel_policy PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(babel_policy PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(babel_policy PRIVATE m)
endif()
